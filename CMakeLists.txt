function(retguard_guard_flags target)
  target_compile_options(${target} PRIVATE -O2 -g -Wall -Wextra -Werror)
endfunction()

# Optional compiler tuning; every function that remains emitted is protected.
# Clang's LLVM threshold is an internal option, so verify support at configure time.
function(retguard_inline_flags output limit)
  if(NOT "${limit}" MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "INLINE_LIMIT must be a positive integer")
  endif()
  if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    set(flags -finline-limit=${limit})
  elseif(CMAKE_C_COMPILER_ID STREQUAL "Clang")
    include(CheckCCompilerFlag)
    include(CheckCXXCompilerFlag)
    check_c_compiler_flag("-mllvm -inline-threshold=${limit}" RETGUARD_C_INLINE_THRESHOLD_${limit})
    check_cxx_compiler_flag("-mllvm -inline-threshold=${limit}" RETGUARD_CXX_INLINE_THRESHOLD_${limit})
    if(NOT RETGUARD_C_INLINE_THRESHOLD_${limit} OR NOT RETGUARD_CXX_INLINE_THRESHOLD_${limit})
      message(FATAL_ERROR "This Clang does not support LLVM inline threshold ${limit}")
    endif()
    # Keep the option and its argument together during CMake option deduplication.
    set(flags "SHELL:-mllvm -inline-threshold=${limit}")
  else()
    message(FATAL_ERROR "INLINE_LIMIT currently requires GCC or Clang")
  endif()
  set(${output} "${flags}" PARENT_SCOPE)
endfunction()

function(retguard_target target)
  # Inherit compiler protection settings unless a caller explicitly selects
  # a mode (for example, the plain/SSP compatibility matrix).
  cmake_parse_arguments(AUTO "" "STACK_FLAG;INLINE_LIMIT" "SOURCES" ${ARGN})
  if(NOT AUTO_SOURCES OR AUTO_UNPARSED_ARGUMENTS OR AUTO_KEYWORDS_MISSING_VALUES)
    message(FATAL_ERROR "retguard_target requires SOURCES and values for any supplied options")
  endif()

  # add_subdirectory consumers call this function from their own directory.
  # Python discovery in the defining directory is not visible in that scope.
  if(NOT Python3_EXECUTABLE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
  endif()

  set(auto_inline_flags)
  if(DEFINED AUTO_INLINE_LIMIT)
    retguard_inline_flags(auto_inline_flags "${AUTO_INLINE_LIMIT}")
  endif()

  find_program(RETGUARD_LLD ld.lld)
  find_program(RETGUARD_GOLD ld.gold)
  if(RETGUARD_LLD)
    set(retguard_linker lld)
  elseif(RETGUARD_GOLD)
    set(retguard_linker gold)
  else()
    message(FATAL_ERROR "retguard_target requires ld.lld or ld.gold for its GAS unwind expressions")
  endif()

  set(assembled_objects)
  set(index 0)
  foreach(source IN LISTS AUTO_SOURCES)
    math(EXPR index "${index} + 1")
    # A one-source object library lets CMake supply the normal C or C++
    # flags to the assembly compiler launcher.
    set(raw_target "${target}_raw_${index}")
    add_library(${raw_target} OBJECT "${source}")
    set_property(TARGET ${raw_target} PROPERTY C_COMPILER_LAUNCHER
      "${Python3_EXECUTABLE};${CMAKE_CURRENT_FUNCTION_LIST_DIR}/tools/retguard_asm_compiler.py")
    set_property(TARGET ${raw_target} PROPERTY CXX_COMPILER_LAUNCHER
      "${Python3_EXECUTABLE};${CMAKE_CURRENT_FUNCTION_LIST_DIR}/tools/retguard_asm_compiler.py")
    set_property(SOURCE "${source}" APPEND PROPERTY OBJECT_DEPENDS
      "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/tools/retguard_asm_compiler.py"
      "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/tools/retguard_coverage.py")
    target_link_libraries(${raw_target} PRIVATE retguard)
    target_compile_definitions(${raw_target} PRIVATE RETGUARD_AUTO)
    set(auto_compiler_flags)
    if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
      list(APPEND auto_compiler_flags -mfentry-name=__retguard_fentry)
    elseif(CMAKE_C_COMPILER_ID STREQUAL "Clang")
      # Clang's generated global constructors miss the return-thunk attribute.
      list(APPEND auto_compiler_flags -mllvm -force-attribute=fn_ret_thunk_extern
        -fno-addrsig)
    endif()
    target_compile_options(${raw_target} PRIVATE -O2 -g -Wall -Wextra -Werror -fPIE
      -pg -mfentry ${auto_compiler_flags}
      -mfunction-return=thunk-extern -fasynchronous-unwind-tables
      ${AUTO_STACK_FLAG} ${auto_inline_flags})

    target_sources(${target} PRIVATE $<TARGET_OBJECTS:${raw_target}>)
    list(APPEND assembled_objects "$<TARGET_OBJECTS:${raw_target}>")
  endforeach()
  target_link_libraries(${target} PRIVATE retguard)
  target_link_options(${target} PRIVATE -fuse-ld=${retguard_linker})
  set_property(TARGET ${target} PROPERTY RETGUARD_PATCHED_OBJECTS "${assembled_objects}")
endfunction()

if(RETGUARD_BUILD_BENCHMARKS)
  find_package(benchmark CONFIG REQUIRED)
  if(BUILD_TESTING)
    add_test(NAME benchmark.test_runner COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/benchmark/test_runner.py)
  endif()
  function(retguard_benchmark name)
    add_executable(${name} benchmark/bench_retguard.cc benchmark/bench_targets.c ${ARGN})
    target_link_libraries(${name} PRIVATE retguard benchmark::benchmark)
    target_compile_options(${name} PRIVATE -O3 -DNDEBUG -fstack-protector-strong)
  endfunction()
  retguard_benchmark(retguard_bench)
  retguard_benchmark(retguard_bench_cet)
  target_compile_options(retguard_bench_cet PRIVATE -fstack-protector-all -fcf-protection=full)
  target_link_options(retguard_bench_cet PRIVATE -Wl,-z,shstk)
  include(CheckCCompilerFlag)
  check_c_compiler_flag(-ffixed-r11 RETGUARD_HAS_FIXED_R11)
  foreach(suffix IN ITEMS gadget gadget_cet)
    retguard_benchmark(retguard_bench_${suffix})
    target_compile_definitions(retguard_bench_${suffix} PRIVATE RETGUARD_GADGET_BENCH)
    if(CMAKE_C_COMPILER_ID STREQUAL "GNU" OR RETGUARD_HAS_FIXED_R11)
      target_compile_options(retguard_bench_${suffix} PRIVATE -ffixed-r11)
      if(CMAKE_C_COMPILER_ID STREQUAL "Clang")
        target_compile_definitions(retguard_bench_${suffix} PRIVATE RETGUARD_CLANG_FIXED_R11)
      endif()
    endif()
  endforeach()
  target_compile_options(retguard_bench_gadget_cet PRIVATE -fstack-protector-all -fcf-protection=full)
  target_link_options(retguard_bench_gadget_cet PRIVATE -Wl,-z,shstk)
  if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    add_executable(retguard_bench_auto benchmark/bench_retguard.cc benchmark/bench_targets.c)
    retguard_target(retguard_bench_auto STACK_FLAG -fstack-protector-strong
      SOURCES benchmark/bench_auto_target.c)
    target_link_libraries(retguard_bench_auto PRIVATE benchmark::benchmark)
    target_compile_definitions(retguard_bench_auto PRIVATE RETGUARD_AUTO_BENCH)
    target_compile_options(retguard_bench_auto PRIVATE -O3 -DNDEBUG -fPIE -fstack-protector-strong)
    target_link_options(retguard_bench_auto PRIVATE -pie)
  endif()
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    set(RETGUARD_BENCH_OPTIMIZATION "O2" CACHE STRING "Library benchmark optimization (O2 or O3)")
    set_property(CACHE RETGUARD_BENCH_OPTIMIZATION PROPERTY STRINGS O2 O3)
    if(NOT RETGUARD_BENCH_OPTIMIZATION MATCHES "^O[23]$")
      message(FATAL_ERROR "RETGUARD_BENCH_OPTIMIZATION must be O2 or O3")
    endif()
    find_package(nlohmann_json CONFIG QUIET)
    find_package(spdlog CONFIG QUIET)
    set(benchmark_libraries stb qoi jsmn sprintf)
    if(nlohmann_json_FOUND)
      list(APPEND benchmark_libraries json)
    else()
      message(STATUS "Skipping nlohmann/json benchmarks: CMake package not found")
    endif()
    if(spdlog_FOUND)
      list(APPEND benchmark_libraries spdlog)
    else()
      message(STATUS "Skipping spdlog benchmarks: CMake package not found")
    endif()
    set(RETGUARD_BENCH_INLINE_LIMIT "200" CACHE STRING
      "Inline benchmark profile setting (GCC inline limit / Clang LLVM threshold)")
    retguard_inline_flags(benchmark_inline_flags "${RETGUARD_BENCH_INLINE_LIMIT}")
    set(library_modes plain auto inline_plain inline_auto)
    foreach(library IN LISTS benchmark_libraries)
      if(library STREQUAL "stb")
        set(driver benchmark/stb_decode.cc)
        set(implementation benchmark/stb_image_impl.c)
      elseif(library STREQUAL "qoi")
        set(driver benchmark/qoi_codec.cc)
        set(implementation benchmark/qoi_impl.c)
      elseif(library STREQUAL "jsmn")
        set(driver benchmark/jsmn_parse.cc)
        set(implementation benchmark/jsmn_impl.c)
      elseif(library STREQUAL "sprintf")
        set(driver benchmark/sprintf_format.cc)
        set(implementation benchmark/sprintf_impl.c)
      elseif(library STREQUAL "json")
        set(driver benchmark/json_parse.cc)
        set(implementation benchmark/json_parse_impl.cc)
      else()
        set(driver benchmark/spdlog_log.cc)
        set(implementation benchmark/spdlog_impl.cc)
      endif()
      foreach(mode IN LISTS library_modes)
        set(target retguard_bench_${library}_${mode})
        add_executable(${target} ${driver})
        target_link_libraries(${target} PRIVATE benchmark::benchmark)
        target_compile_options(${target} PRIVATE -${RETGUARD_BENCH_OPTIMIZATION}
          -DNDEBUG -fPIE -fstack-protector-strong)
        target_link_options(${target} PRIVATE -pie)
        set(inline_arguments)
        if(mode MATCHES "^inline_")
          list(APPEND inline_arguments INLINE_LIMIT ${RETGUARD_BENCH_INLINE_LIMIT})
        endif()
        if(mode MATCHES "auto$")
          retguard_target(${target} STACK_FLAG -fstack-protector-strong
            ${inline_arguments} SOURCES ${implementation})
          set(impl ${target}_raw_1)
        else()
          set(impl ${target}_impl)
          add_library(${impl} OBJECT ${implementation})
          target_compile_options(${impl} PRIVATE -O2 -g -Wall -Wextra -Werror
            -fPIE -fstack-protector-strong -fno-omit-frame-pointer -fno-optimize-sibling-calls
            -fasynchronous-unwind-tables)
          target_sources(${target} PRIVATE $<TARGET_OBJECTS:${impl}>)
        endif()
        # Appended after helper flags so both implementations use the selected
        # optimization and identical inlining policy.
        target_compile_options(${impl} PRIVATE -${RETGUARD_BENCH_OPTIMIZATION} -DNDEBUG)
        if(mode STREQUAL "inline_plain")
          target_compile_options(${impl} PRIVATE ${benchmark_inline_flags})
        endif()
        if(library STREQUAL "stb")
          target_compile_definitions(${target} PRIVATE
            RETGUARD_PNG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/benchmark/data/sample.png")
          target_include_directories(${impl} PRIVATE third_party/stb)
          target_compile_options(${impl} PRIVATE -Wno-unused-function)
        elseif(library STREQUAL "qoi")
          target_include_directories(${impl} PRIVATE third_party/qoi)
        elseif(library STREQUAL "jsmn")
          target_include_directories(${impl} PRIVATE third_party/jsmn)
        elseif(library STREQUAL "sprintf")
          target_include_directories(${impl} SYSTEM PRIVATE third_party/stb)
        elseif(library STREQUAL "json")
          target_link_libraries(${impl} PRIVATE nlohmann_json::nlohmann_json)
          target_link_libraries(${target} PRIVATE nlohmann_json::nlohmann_json)
        else()
          target_link_libraries(${impl} PRIVATE spdlog::spdlog_header_only)
          target_link_libraries(${target} PRIVATE spdlog::spdlog_header_only)
        endif()
        if(BUILD_TESTING)
          add_test(NAME benchmark.${library}.${mode}.correctness COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/benchmark/check_correctness.py $<TARGET_FILE:${target}>)
        endif()
      endforeach()
    endforeach()
  endif()
endif()
